def parse_cookies(cookies_json: str) -> list:
    """Convert EditThisCookie JSON → Playwright cookie format."""
    try:
        raw = json.loads(cookies_json.strip())
    except (json.JSONDecodeError, ValueError):
        return []

    same_site_map = {
        "no_restriction": "None",
        "lax":            "Lax",
        "strict":         "Strict",
        "unspecified":    "None",
    }

    result = []
    for c in raw:
        raw_ss   = str(c.get("sameSite", "no_restriction")).lower()
        cookie   = {
            "name":     c.get("name", ""),
            "value":    c.get("value", ""),
            "domain":   c.get("domain", ""),
            "path":     c.get("path", "/"),
            "secure":   bool(c.get("secure", False)),
            "httpOnly": bool(c.get("httpOnly", False)),
            "sameSite": same_site_map.get(raw_ss, "None"),
        }
        if "expirationDate" in c:
            cookie["expires"] = int(c["expirationDate"])
        result.append(cookie)
    return result


def is_login_page(page) -> bool:
    """Return True if the current page is an Intel/Azure B2C login page."""
    url = page.url.lower()
    return (
        "login" in url
        or "consumer.intel.com" in url
        or "consumerint.intel.com" in url
        or "b2c_1a_unifiedlogin" in url
    )


def inject_cookies_and_reload(page, context, cookies: list) -> bool:
    """Inject cookies into the browser context and reload the page."""
    if not cookies:
        return False

    # Must be on the target domain before adding cookies
    domain_root = BASE_URL.rstrip("/")
    if domain_root not in page.url:
        page.goto(domain_root, wait_until="domcontentloaded", timeout=30000)
        time.sleep(2)

    try:
        context.add_cookies(cookies)
        print(f"  [OK] Injected {len(cookies)} cookie(s).")
    except Exception as e:
        print(f"  [WARN] Cookie injection partial error: {e}")

    # Reload so cookies take effect
    page.reload(wait_until="domcontentloaded", timeout=30000)
    time.sleep(3)
    return True


def do_sso_login_form(page) -> bool:
    """
    Attempt the standard Azure B2C email → password form login.
    This may be blocked by Intel's bot detection, but we try anyway
    as a fallback when no cookies are provided.
    """
    print("  [INFO] Attempting SSO form login (may be blocked by bot detection)...")
    print(f"  Login URL: {page.url}")

    # Email
    for sel in ["input[name='loginfmt']", "input[id='i0116']", "input[type='email']"]:
        try:
            page.wait_for_selector(sel, timeout=8000, state="visible")
            page.fill(sel, EMAIL)
            print(f"  [OK] Email entered  ({sel})")
            break
        except PWTimeout:
            continue
    else:
        print("  [ERR] Email field not found.")
        page.screenshot(path="login_email_debug.png")
        return False

    time.sleep(0.6)

    # Next
    for sel in ["#idSIButton9", "button[type='submit']", "input[type='submit']",
                "input[value='Next']", "button:has-text('Next')"]:
        try:
            page.click(sel, timeout=5000)
            print(f"  [OK] Next clicked  ({sel})")
            break
        except PWTimeout:
            continue

    time.sleep(3)

    # Password
    for sel in ["input[name='passwd']", "input[id='i0118']", "input[type='password']"]:
        try:
            page.wait_for_selector(sel, timeout=10000, state="visible")
            page.fill(sel, PASSWORD)
            print(f"  [OK] Password entered  ({sel})")
            break
        except PWTimeout:
            continue
    else:
        print("  [ERR] Password field not found.")
        page.screenshot(path="login_pwd_debug.png")
        return False

    time.sleep(0.6)

    # Sign in
    for sel in ["#idSIButton9", "button[type='submit']", "input[type='submit']",
                "input[value='Sign in']", "button:has-text('Sign in')"]:
        try:
            page.click(sel, timeout=5000)
            print(f"  [OK] Sign in clicked")
            break
        except PWTimeout:
            continue

    time.sleep(4)

    # Stay signed in prompt
    for sel in ["#idSIButton9", "input[value='Yes']", "#idBtn_Back"]:
        try:
            page.click(sel, timeout=5000)
            print(f"  [OK] Stay-signed-in prompt handled")
            time.sleep(2)
            break
        except PWTimeout:
            continue

    # Poll for redirect back to app
    print("  Waiting for redirect (up to 90s)...")
    print("  ", end="", flush=True)
    for i in range(90):
        time.sleep(1)
        print(".", end="", flush=True)
        cur = page.url
        # Success: back on the app domain
        if BASE_URL.replace("https://", "").rstrip("/") in cur and "login" not in cur.lower():
            print(f"\n  [OK] Redirected after {i+1}s → {cur}")
            return True
        # Hard failure: Intel bot-block page
        try:
            body = page.inner_text("body") or ""
            if ("Please provide the following details" in body
                    and "contact administrator" in body.lower()):
                print(f"\n  [ERR] Intel Azure B2C bot-block detected.")
                print("        Add cookies to COOKIES_JSON and re-run.")
                page.screenshot(path="login_blocked.png")
                return False
        except Exception:
            pass

    print()
    # Final fallback: navigate directly (SAML cookie may already be set)
    print("  [INFO] Timeout — navigating to app directly...")
    page.goto(BASE_URL, wait_until="domcontentloaded", timeout=30000)
    time.sleep(4)
    return BASE_URL.replace("https://", "").rstrip("/") in page.url

                    print("  [OK] ✓ Logged in via cookies!")
                    logged_in = True

            else:
                # ── FALLBACK: form login ──────────────────────
                logged_in = do_sso_login_form(page)

            if not logged_in:
                print("❌ Login failed. Cannot continue.")
                page.screenshot(path="04_login_failed.png")
                browser.close()
                return

        else:
            print("  [OK] Already authenticated — no login needed.")
            logged_in = True

        print(f"  Current URL after login: {page.url}")
        page.screenshot(path="04_logged_in.png")

        # Navigate back to app if login redirected elsewhere
        if BASE_URL.replace("https://", "").rstrip("/") not in page.url:
            print(f"  Navigating back to {BASE_URL}...")
            page.goto(BASE_URL, wait_until="domcontentloaded", timeout=30000)
            time.sleep(3)
            page.screenshot(path="04b_back_at_app.png")

        # ── STEP 5: Click Create a Project ───────────────────
        print("➡ Clicking Create a Project...")
        time.sleep(3)   # let page fully render after login

        try:
            page.wait_for_selector("button:has-text('Create a Project')", timeout=30000)
            page.click("button:has-text('Create a Project')")
            print("  [OK] 'Create a Project' clicked.")
        except PWTimeout:
            print("  [WARN] 'Create a Project' button not found via primary selector.")
            print("         Trying alternative selectors...")

            # Try alternatives
            alternatives = [
                "text=Create a Project",
                "a:has-text('Create a Project')",
                "[aria-label*='Create a Project']",
                "xpath=//button[contains(.,'Create a Project')]",
                "xpath=//a[contains(.,'Create a Project')]",
            ]
            clicked = False
            for sel in alternatives:
                try:
                    page.wait_for_selector(sel, timeout=5000, state="visible")
                    page.click(sel)
                    print(f"  [OK] Clicked via: {sel}")
                    clicked = True
                    break
                except PWTimeout:
                    continue

            if not clicked:
                # Dump visible buttons to help debug
                print("  [DEBUG] Visible buttons on current page:")
                try:
                    btns = page.query_selector_all("button, a[role='button']")
                    for b in btns[:25]:
                        try:
                            t = b.inner_text().strip()
                            if t:
                                print(f"    • '{t}'")
                        except Exception:
                            pass
                except Exception:
                    pass
                page.screenshot(path="05_create_project_debug.png")
                print("  [📷] 05_create_project_debug.png")

        time.sleep(2)
        page.screenshot(path="05_create_project.png")
        print("  [📷] 05_create_project.png")

        # ── STEP 6: Validate Project Page Sections ────────────
        print("➡ Validating Project Page Sections...")

        page.wait_for_selector("text=Overview", timeout=20000)

        sections = [
            "Overview",
            "Details",
            "Additional Info",
            "Application Steps",
            "Media",
        ]

        all_found = True
        for section in sections:
            try:
                page.wait_for_selector(f"text={section}", timeout=20000)
                print(f"  ✅ Found section: {section}")
            except PWTimeout:
                print(f"  ❌ Section NOT found: {section}")
                all_found = False

        # Screenshot
        page.screenshot(path="project_page.png")
        print("  [📷] project_page.png")

        if all_found:
            print("\n🎉 SUCCESS: Project page loaded with all sections!")
        else:
            print("\n⚠  PARTIAL: Some sections were not found — check screenshots.")

        browser.close()
        print("\nBrowser closed. Done.")
